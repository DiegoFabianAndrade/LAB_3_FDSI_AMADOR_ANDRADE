--------------------------------------------------------------------

2. MARCA DE TIEMPO UTC DE INICIO
--------------------------------------------------------------------

3. PRUEBAS RED TEAM (EJECUTAR EN POWERSHELL)
--------------------------------------------------------------------
# A. Peticion HTTP a la pagina principal
curl.exe -i http://127.0.0.1/

# B. Obtencion de cabeceras HTTP de seguridad
curl.exe -I http://127.0.0.1/

# C. Consulta del inventario publico de activos
curl.exe -i http://127.0.0.1/public-inventory.txt

# D. Rafaga de reconocimiento de rutas (Prueba de error 404)
curl.exe -s -o NUL -w "%{http_code} /.git/config`n" http://127.0.0.1/.git/config
curl.exe -s -o NUL -w "%{http_code} /.env`n" http://127.0.0.1/.env
curl.exe -s -o NUL -w "%{http_code} /wp-config.php`n" http://127.0.0.1/wp-config.php
curl.exe -s -o NUL -w "%{http_code} /admin`n" http://127.0.0.1/admin
curl.exe -s -o NUL -w "%{http_code} /backup.sql`n" http://127.0.0.1/backup.sql
